```
export default function ThreadList({ fetchRuns = listRunsV2, pageStep = 20, maxRows = 100, ... }) {
  const [runs, setRuns]       = useState([])      // 全量窗口（created_at 降序——后端已排序，前端不重排）
  const [limit, setLimit]     = useState(pageStep)
  const [loadState, setLoadState] = useState('ok') // 'ok'|'error'——失败≠空库（UI-14 同位纪律）
  const [query, setQuery]     = useState('')       // 搜索态不变（L2 维持现状）

  load(limitValue) {                       // 初载与「加载更早」共用
    setLoadState('loading')
    try { setRuns(await fetchRuns(limitValue)); setLoadState('ok') }
    catch { setLoadState('error') }        // 保留现计数行撤下+「暂无数据」失败空态
  }
  useEffect(() => { load(pageStep) }, [])  // 初始 limit=20（规格第 14 行）
  loadMore() {                            // 语义不变：搜索中/到顶不触发
    if (searching || atEarliest) return
    const next = Math.min(limit + pageStep, maxRows)
    setLimit(next); load(next)             // 扩大 limit 重拉全量替换（无 offset 的既定映射）
  }
  // 行渲染/搜索过滤/选中高亮——现状代码不动
}
```
