## 1. 施工伪代码

### 1.1 graph.py —— 渲染单点

```python
_PROCESS_NOTES_BUDGET = 400          # 方案 §3.4 有界值
_PROCESS_NOTES_HEADER = "【本 run 已做（过程档案摘要——防重参考，不是作答材料）】"

def _render_process_notes(state, *, budget=_PROCESS_NOTES_BUDGET) -> str:
    if not state.process_notes:
        return ""                                    # 零档案＝零段（首轮零噪声）
    lines, total = [], len(_PROCESS_NOTES_HEADER)
    for note in state.process_notes:                 # 逐条追加，超预算即止
        line = f"- [{phase}] {title}｜{digest}"
        if total + len(line) + 1 > budget:
            lines.append(f"-（过程档案摘要有界截断——已渲染 {len(lines)}/{总数} 条）")
            break                                    # 截断必披露（不静默）
        lines.append(line); total += len(line) + 1
    return header + "\n" + "\n".join(lines)
```
