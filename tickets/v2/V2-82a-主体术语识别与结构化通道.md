```python
def subject_candidates(question: str, dump: dict) -> tuple[str, ...]:
    """题面主体候选（油田/区块/井/平台名归一形，保序去重）。

    复用 anchors.entity_anchor_tokens 抽实体 token → glossary_aliases 归一 →
    well_ids.normalize_well_id 判井号形（返空串则非井号）。
    只从题面提取，绝不推测题面没有的主体。
    """

def term_candidates(question: str, dump: dict) -> tuple[str, ...]:
    """题面术语候选（台账指标名/术语名及其别名，保序去重）。

    v1 essence 重写：NFKC+casefold+去空白双侧归一，needle=归一后题面；
    对 dump["metrics"] 各名+别名、dump["terms"] 各 term+别名逐个子串包含判定，
    命中收规范名。注意 roster.metric_roster 不含术语正名——必须直读 dump。
    """
```
