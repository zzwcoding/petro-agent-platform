## 施工基准（伪代码——照抄，偏离即停报 L0）

```python
# context/materials.py
MAX_REPORT_SEATS_PER_SOURCE = 6   # 单来源席位上界（本块常量点；数值不落规格）

def _select_reports(reports, cap, anchors):
    # reports：信任级=report 的观察（保持原序）；anchors：锚点串（见下）
    by_source: dict[str, list[obs]] = OrderedDict()      # 来源序=首次出现序
    for obs in reports:
        by_source.setdefault(source_of(obs), []).append(obs)
    chosen: list[obs] = []
    chosen_keys: set[int] = set()                         # 用 id(obs) 或 index
    source_count: Counter[str] = Counter()

    def _take(obs) -> bool:
        key = int(obs["index"])
        src = source_of(obs)
        if key in chosen_keys or len(chosen) >= cap:
            return False
        if source_count[src] >= MAX_REPORT_SEATS_PER_SOURCE:
            return False
        chosen.append(obs); chosen_keys.add(key); source_count[src] += 1
        return True

    # ① 锚席位：按 anchors 顺序，为每个锚取首个「内容或来源含锚」且尚未入选的片段
    for anchor in anchors:
        for obs in reports:
            if anchor and (anchor in str(obs.get("content") or "")
                           or anchor in source_of(obs)):
                if _take(obs):
                    break
    # ② 来源保底席：每个来源首个未选片段（来源序）
    for src, rows in by_source.items():
        for obs in rows:
            if int(obs["index"]) not in chosen_keys:
                _take(obs); break
    # ③ 轮转补席：按来源轮转取未选片段，直到 cap
    cursor = {src: 0 for src in by_source}
    while len(chosen) < cap:
        progressed = False
        for src, rows in by_source.items():
            while cursor[src] < len(rows):
                obs = rows[cursor[src]]; cursor[src] += 1
                if _take(obs):
                    progressed = True
                    break
        if not progressed:
            break
    return chosen

# render_materials 签名（additive）：
def render_materials(observations, *, max_observations, anchors: Sequence[str] = ()):
    material, authority, reports = _carrier_split(observations)
    truncated = len(reports) > max_observations
    ordered = authority + _select_reports(reports, max_observations, anchors)
    # 其余（label/content/血缘口径行/披露行/元数据）逐字不变
```

- **锚点来源**（answering 调用点）：`anchors = tuple(state.subjects) + tuple(state.terms) + _numeric_anchor_tokens(state.question)`；
  `_numeric_anchor_tokens`＝纯字符扫描取连续数字串（含小数/百分号/千分位形态；`re` 禁用——形态处理，先例 `understanding/closure.numeric_form_hit`），去重保序，最多 8 个。
- **不改**：披露行逐字、`material_truncated/rendered/total` 三字段、`selected_reason` 文本、[n] 序号（用观察自带 `index`）、权威族全量。
- 输出块顺序＝权威原序＋选中检索片段**按 index 升序**（可读性；不影响引标）。
