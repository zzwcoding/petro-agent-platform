### ① `clarification/guidance.py`

```python
# 换链/未绑定闭世界（结构化判据——唯一来源 preconditions.public 常量）
_REROUTE_RULE_IDS = frozenset({HIT_RULE_CHAIN_SWITCH, HIT_RULE_UNREGISTERED_METRIC})

def _has_binding_refusal(state):
    for verdict in state.precondition_verdicts:
        if not isinstance(verdict, Mapping): continue
        if verdict.get("decision") != "refuse": continue
        reason = verdict.get("reason")
        if isinstance(reason, Mapping) and reason.get("rule_id") in _REROUTE_RULE_IDS:
            return True
    return False
```
