```python
# 1) 解析（parsing 新件）：作答第二出口
def parse_explore(raw) -> Explore | None:
    obj = try_parse_whole_json(raw)                 # 复用整枚 JSON 判定
    if not obj or obj.get("action") != "explore": return None
    reason = str(obj.get("reason") or "").strip()
    reqs = obj.get("requests") or []
    if not reason or not (1 <= len(reqs) <= 2): return None
    for r in reqs:
        tool = r.get("tool")
        if tool not in {"doc_retrieve","fetch_section","fetch_report"}: return None
        if tool == "doc_retrieve" and not str(r.get("query") or "").strip(): return None
        if tool in {"fetch_section","fetch_report"} and not str(r.get("report_id") or "").strip(): return None
    return Explore(reason=reason, requests=reqs)

# 2) 作答出口：先探后答
#    answer 调用后：draft = extract_finalize_answer(raw) or raw
#    若 parse_explore(draft) 命中 → 不落答案，落 next_intent="explore"
#    否则照现有剥离/起草/核验管线

# 3) 编排：explore 执行
def run_explore(state, exp):
    seen = {o["report_id"] for o in state.observations if o.get("report_id")}
    for r in exp.requests:
        fp = fingerprint(r)
        if fp in state.explore_fingerprints: reject("重复请求", r); continue
        if r.tool in {"fetch_section","fetch_report"} and r.report_id not in seen:
            reject("只能对已见来源要更多", r); continue
        obs = dispatch(r)          # doc_retrieve→L0/L1；fetch_*→KB 读（截断+披露）
        state.observations += obs; state.explore_fingerprints.append(fp)
    append_message(state, role="user", content=explore_increment(state, exp))   # 头部+新料（[n]续排）
    state.explore_rounds += 1
    state.next_intent = "answer"   # 回作答

# 4) 路由与硬停止
#    answer → next_intent=="explore" 且 explore_rounds < 2 → run_explore → answer
#    若 explore_rounds >= 2 仍 explore → 强制作答（提示：只能如实作答，列已试/仍缺）

# 5) 新工具
#    fetch_section(report_id, section=None) → KB 取该节片段（或代表节）；受 fetch_cap
#    fetch_report(report_id, max_chars)     → kb_report.doc 截断+披露；最后手段
```
